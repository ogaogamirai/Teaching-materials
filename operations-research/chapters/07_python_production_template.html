🐍 7. 【実践】Python (Google OR-Tools) で書く黄金の型

世界中の現場で使われている、Google OR-Tools CP-SATの実装コードです。どんな問題もこの「黄金の3ステップ」で解けます。

from ortools.sat.python import cp_model

# 0. 基本データ（スタッフ・曜日）
staff = ["Alice", "Bob", "Charlie", "Dave"]
days = ["月", "火", "水", "木", "金", "土", "日"]

# 1. モデルの初期化
model = cp_model.CpModel()

# --- ステップ①：変数を置く（何を決めるのか？） ---
# x[スタッフ, 日, シフト] in {0, 1}
x = {}
for s in staff:
    for d in days:
        for shift in ["D", "N", "O"]:
            x[s, d, shift] = model.NewBoolVar(f"x_{s}_{d}_{shift}")

# --- ステップ②：ハード制約を敷く（鉄の掟） ---
# [C1] 1人1日1シフトのみ（日・夜・休の排他）
for s in staff:
    for d in days:
        model.AddExactlyOne([x[s, d, shift] for shift in ["D", "N", "O"]])

# [C2] 毎日、日勤1名、夜勤1名
for d in days:
    model.Add(sum(x[s, d, "D"] for s in staff) == 1)
    model.Add(sum(x[s, d, "N"] for s in staff) == 1)

# [C3] 夜勤の翌日は日勤禁止！（足して1以下）
for s in staff:
    for d in range(len(days) - 1):
        model.Add(x[s, days[d], "N"] + x[s, days[d+1], "D"] <= 1)

# [C4] どのスタッフにも一週間に最低1回の2連休！（補助変数＋AND＋OR）
for s in staff:
    two_off = []
    for d in range(len(days) - 1):
        pair_flag = model.NewBoolVar(f"two_off_{s}_{d}")
        model.AddBoolAnd([x[s, days[d], "O"], x[s, days[d+1], "O"]]).OnlyEnforceIf(pair_flag)
        two_off.append(pair_flag)
    model.Add(sum(two_off) >= 1)

# [CF] 公平性：夜勤は1人最大2回まで（テンプレート制約5）
for s in staff:
    model.Add(sum(x[s, d, "N"] for d in days) <= 2)

# --- ステップ③：ソフト制約（目的関数：ペナルティ最小化） ---
penalties = []
penalties.append(10 * (1 - x["Alice", "水", "O"])) # Alice水曜休みの希望
penalties.append(10 * (1 - x["Bob", "土", "O"])) # Bob土曜休みの希望
penalties.append(10 * (1 - x["Bob", "日", "O"])) # Bob日曜休みの希望
penalties.append(10 * (1 - x["Charlie", "月", "O"])) # Charlie月曜休みの希望
penalties.append(10 * (1 - x["Dave", "金", "N"])) # Dave金曜夜勤の希望
model.Minimize(sum(penalties))

# 実行！
solver = cp_model.CpSolver()
status = solver.Solve(model)
